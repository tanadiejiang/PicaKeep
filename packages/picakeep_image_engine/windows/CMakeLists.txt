cmake_minimum_required(VERSION 3.22)
project(picakeep_image_engine_windows LANGUAGES C CXX)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../native" "${CMAKE_CURRENT_BINARY_DIR}/core")
add_library(picakeep_image_engine_plugin SHARED picakeep_image_engine_plugin_c_api.cpp)
target_compile_features(picakeep_image_engine_plugin PUBLIC cxx_std_17)
target_compile_definitions(picakeep_image_engine_plugin PRIVATE FLUTTER_PLUGIN_IMPL)
target_include_directories(picakeep_image_engine_plugin INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(picakeep_image_engine_plugin PRIVATE flutter flutter_wrapper_plugin picakeep_image_engine)
set(picakeep_image_engine_bundled_libraries "$<TARGET_FILE:picakeep_image_engine>" PARENT_SCOPE)
